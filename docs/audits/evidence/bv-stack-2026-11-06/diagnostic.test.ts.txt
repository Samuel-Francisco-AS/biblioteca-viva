import { writeFileSync } from 'node:fs';
import { it, expect } from 'vitest';
import { Mesh, InstancedMesh, Group } from 'three';
import { performance } from 'node:perf_hooks';
import { CameraNavigation } from './src/features/library/three/CameraNavigation';
import { OrthographicCamera } from 'three';
import { LIBRARY_ROOMS } from './src/features/library/libraryBuildingGeometry';
import { LIBRARY_SHELVES } from './src/features/library/libraryShelfLayout';
import { createProceduralLibraryBuilding } from './src/features/library/three/proceduralLibraryBuilding';
import { createProceduralComposition } from './src/features/library/three/proceduralComposition';
import { collectInhabitantObstacles } from './src/features/library/three/inhabitantObstacles';
import { InhabitantNavigation } from './src/features/library/inhabitantNavigation';
import { disposeObjectTree } from './src/features/library/three/referenceScene';
it('audit structural inventory and bounded CPU diagnostic; no GPU',()=>{
 const root = new Group(); root.add(createProceduralLibraryBuilding(true), createProceduralComposition(LIBRARY_SHELVES).root);
 const geometries=new Set(), materials=new Set(), textures=new Set(); let objects=0,meshes=0,instances=0,triangles=0,vertices=0,transparent=0;
 root.traverse(o=>{objects++; if(o instanceof Mesh){ meshes++;const n=o instanceof InstancedMesh?o.count:1;instances+=n;geometries.add(o.geometry);vertices+=(o.geometry.attributes.position?.count??0)*n;triangles+=(o.geometry.index?.count??o.geometry.attributes.position?.count??0)/3*n;for(const m of Array.isArray(o.material)?o.material:[o.material]){materials.add(m);if(m.transparent)transparent++;for(const v of Object.values(m))if(v?.isTexture) textures.add(v);}} });
 const t=performance.now(); const obstacles=collectInhabitantObstacles(root.children);const collectMs=performance.now()-t;
 const nav=new InhabitantNavigation(obstacles); const start=performance.now();nav.route({x:0,z:0},{x:-12,z:6},0.45);const coldRouteMs=performance.now()-start;
 const camera=new CameraNavigation(new OrthographicCamera(),1,LIBRARY_ROOMS,10); camera.setViewport(412,840);
 const times=[];for(let i=0;i<300;i++){const t=performance.now();camera.panBetweenScreenPoints({x:200,y:400},{x:201+(i%2),y:401}); times.push(performance.now()-t);}times.sort((a,b)=>a-b);
 writeFileSync('/tmp/bv-audit-worktree-metrics.json',JSON.stringify({scope:'building+shelves only; all instances including culled; no renderer',objects,meshes,instances,triangles,vertices,geometries:geometries.size,materials:materials.size,textures:textures.size,transparent,obstacles:obstacles.length,nodes:nav.nodes.length,collectMs,coldRouteMs,panMedianMs:times[150],panP95Ms:times[285],panMaxMs:times[299]}));
 expect(meshes).toBeGreaterThan(0);expect(times.every(Number.isFinite)).toBe(true);disposeObjectTree(root);
});
